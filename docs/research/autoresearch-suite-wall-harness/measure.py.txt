ROOT = Path(__file__).resolve().parents[2]
EVIDENCE = Path('C:/x/autoresearch-suite-wall')

CARD = 't_82b1bc95'
NAME = sys.argv[1]
if not NAME or any(c not in 'abcdefghijklmnopqrstuvwxyz0123456789-' for c in NAME):
    raise SystemExit('invalid run name')
slot = runpy.run_path(str(SLOT), run_name='slot_probe')

def holders():
    result = []
    for path in Path(slot['TMP']).glob('agr-suite-slot-*.lock'):
        try:
            row = json.loads(path.read_text(encoding='utf-8'))
        except (OSError, ValueError):
            raise RuntimeError('unreadable holder: ' + str(path))
        if slot['pid_alive'](int(row.get('pid',0))):
            result.append(row)
    return result

def merges():
    script = "Get-CimInstance Win32_Process | Where-Object { $_.Name -match 'python' -and $_.CommandLine -match 'merge_(batch|chain)\\.py' } | Select-Object ProcessId | ConvertTo-Json -Compress"
    p = subprocess.run(['powershell.exe','-NoProfile','-Command',script],capture_output=True,text=True,timeout=45)
    if p.returncode:
        raise RuntimeError('merge-process query failed: '+p.stderr)
    return json.loads(p.stdout) if p.stdout.strip() else []

def cpu_times():
    idle,kernel,user = (ctypes.c_ulonglong() for _ in range(3))
    if not ctypes.windll.kernel32.GetSystemTimes(ctypes.byref(idle),ctypes.byref(kernel),ctypes.byref(user)):
        raise RuntimeError('GetSystemTimes failed')
    return idle.value,kernel.value+user.value

def head():
    return subprocess.check_output(['git','rev-parse','HEAD'],cwd=ROOT,text=True).strip()

def tracked():
    data = subprocess.check_output(['git','diff','HEAD','--binary'],cwd=ROOT)
    others = subprocess.check_output(['git','ls-files','--others','--exclude-standard','-z'],cwd=ROOT)
    data += others
    for raw in others.split(b'\0'):
        if raw:
            path = ROOT / os.fsdecode(raw)
            if path.is_file():
                data += hashlib.sha256(path.read_bytes()).digest()
    return hashlib.sha256(data).hexdigest()

def emit(path, row):
    with path.open('x',encoding='utf-8') as f:
        json.dump(row,f,indent=2,allow_nan=False)

EVIDENCE.mkdir(parents=True,exist_ok=True)
base = EVIDENCE / NAME
if '--inside' not in sys.argv:
    sys.stdout = base.with_suffix('.runner.log').open('x', encoding='utf-8', buffering=1)
    sys.stderr = sys.stdout
    # Fail closed on resume/stale output, including a second waiter with the same name.
    claim = base.with_suffix('.claim.json')
    emit(claim,{'pid':os.getpid(),'root':str(ROOT),'head':head(),'created':time.time()})
    print('WAITING',os.getpid(),str(base.with_suffix('.json')),flush=True)
    wait_start=time.monotonic()
    while True:
        live=holders()
        merge=merges()
        if not live and not merge:
            a=cpu_times(); time.sleep(20); b=cpu_times()
            total=b[1]-a[1]
            load=100*(1-(b[0]-a[0])/total) if total>0 else 100
            if load<=10 and not holders() and not merges():
                break
            print('WAIT idle CPU',load,flush=True)
        else:
            print('WAIT drain',live,merge,flush=True)
            time.sleep(30)
    emit(base.with_suffix('.launch.json'),{'pid':os.getpid(),'head':head(),'wait_s':time.monotonic()-wait_start,'idle_cpu_percent':load})
    argv=[sys.executable,str(SLOT),'--card',CARD,'--',sys.executable,str(Path(__file__).resolve()),NAME,'--inside']
    raise SystemExit(subprocess.call(argv,cwd=ROOT,stdout=sys.stdout,stderr=subprocess.STDOUT))

try:
